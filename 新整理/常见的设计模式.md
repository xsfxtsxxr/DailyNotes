### 1.工厂模式

```js
var FactoryFunc = function (role) {
  function Admin() {
    this.name = '管理员'
    this.permission = ['首页', '查询', '权限管理']
  }
  function User() {
    this.name = '普通用户'
    this.permission = ['首页', '查询']
  }
  switch (role) {
    case 'admin':
      return new Admin()
    case 'user':
      return new User()
    default:
      throw new Error('参数错误, 可选参数: admin、user')
  }
}
var admin = FactoryFunc('admin')
var user = FactoryFunc('user')
```

### 2.单例模式

Vuex 的 Store 在实例化时，就算你实例化多个，也只存在一个 Store

```js
function SingleObject() {
  this.instance = null
  this.name = '单例'
}
SingleObject.getInstance = function () {
  if (this.instance === null) {
    this.instance = new SingleObject()
  } else {
    return this.instance
  }
}
var obj1 = SingleObject.getInstance()
var obj2 = SingleObject.getInstance()

console.log(obj1 === obj2)
```

```js
class SalesOffice {
  constructor() {
    //缓存列表(售楼花名册)
    this.messageList = []
  }
  //订阅函数(订阅售楼消息)
  listen(fn) {
    this.messageList.push(fn)
  }
  //发布函数(发布售楼消息)
  pop(){
    for(let i=0;i<this.messageList.length;i++>){
      var fn = this.messageList[i]
      fn.apply(this,arguments)
    }
  }
}
var salesOffice = new SalesOffice()
//小明订阅售楼消息
salesOffice.listen(function(msg){
  console.log('小明'+msg)
})
//小红订阅售楼消息
salesOffice.listen(function(msg){
  console.log('小红'+msg)
})
//售楼处发布售楼消息
salesOffice.pop('你好，今天的房价为2万一平！')
//打印结果:
// 小明你好，今天的房价为2万一平！
// 小红你好，今天的房价为2万一平！
```

### 4.代理模式

```js
var obj = {
  value: 1
}
new Proxy(obj, function () {
  get(target,property){
    if (property in target) {
      return target[property]
    } else {
      throw new ReferenceError("Property \"" + property + "\" does not exist.")
    }
  }
})
```
